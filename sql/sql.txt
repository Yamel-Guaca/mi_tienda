CREATE TABLE IF NOT EXISTS `corresponsal_closures` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  `closure_date` DATE NOT NULL UNIQUE,
  `branch_id` INT NULL,
  `user_id` INT NULL,
  `total_coins` DECIMAL(15,2) NOT NULL DEFAULT 0,
  `total_bills` DECIMAL(15,2) NOT NULL DEFAULT 0,
  `total_cash` DECIMAL(15,2) NOT NULL DEFAULT 0,
  `total_platforms` DECIMAL(15,2) NOT NULL DEFAULT 0,
  `subtotal_cash_platforms` DECIMAL(15,2) NOT NULL DEFAULT 0,
  `total_system` DECIMAL(15,2) NOT NULL DEFAULT 0,
  `grand_total` DECIMAL(15,2) NOT NULL DEFAULT 0,
  `base_snapshot` DECIMAL(15,2) NOT NULL DEFAULT 10000000.00,
  `recaudo` DECIMAL(15,2) NOT NULL DEFAULT 0,
  `transactions_count` INT NOT NULL DEFAULT 0,
  `notes` VARCHAR(255) NULL,
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 4. Valores ingresados por cada columna dinámica en cada cierre
CREATE TABLE IF NOT EXISTS `corresponsal_closure_values` (
  `closure_id` INT NOT NULL,
  `column_id` INT NOT NULL,
  `amount` DECIMAL(15,2) NOT NULL DEFAULT 0,
  PRIMARY KEY (`closure_id`, `column_id`),
  CONSTRAINT `fk_ccv_closure` FOREIGN KEY (`closure_id`) REFERENCES `corresponsal_closures` (`id`) ON DELETE CASCADE,
  CONSTRAINT `fk_ccv_column` FOREIGN KEY (`column_id`) REFERENCES `corresponsal_columns` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- 5. Arqueo detallado de monedas y billetes (Auditoría)
CREATE TABLE IF NOT EXISTS `corresponsal_cash_counts` (

  `subtotal` DECIMAL(15,2) NOT NULL DEFAULT 0,
  PRIMARY KEY (`closure_id`, `denomination`, `cash_type`),
  CONSTRAINT `fk_ccc_closure` FOREIGN KEY (`closure_id`) REFERENCES `corresponsal_closures` (`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
